import arcade

# --------------------
# НАСТРОЙКИ
# --------------------
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
SCREEN_TITLE = "Platformer – Game Over Demo"

PLAYER_SPEED = 5
JUMP_SPEED = 14
GRAVITY = 0.45

LEVEL_WIDTH = 2200

ENEMY_SPEED = 2
ATTACK_TIME = 0.15

PLAYER_MAX_HP = 10
INVINCIBLE_TIME = 1.0


# --------------------
# УТИЛИТА СПРАЙТОВ
# --------------------
def make_sprite(width, height, color):
    texture = arcade.make_soft_square_texture(
        size=max(width, height),
        color=color,
        outer_alpha=255,
        center_alpha=255
    )
    sprite = arcade.Sprite()
    sprite.texture = texture
    sprite.width = width
    sprite.height = height
    return sprite


# --------------------
# ИГРОК
# --------------------
class Player(arcade.Sprite):
    def __init__(self):
        super().__init__()
        self.texture = arcade.make_soft_square_texture(
            64, arcade.color.BLUE, 255, 255
        )
        self.width = 40
        self.height = 60

        self.facing = 1
        self.hp = PLAYER_MAX_HP
        self.invincible_timer = 0


# --------------------
# ВРАГ
# --------------------
class Enemy(arcade.Sprite):
    def __init__(self, x, y, platforms):
        super().__init__()
        self.texture = arcade.make_soft_square_texture(
            48, arcade.color.RED, 255, 255
        )
        self.width = 40
        self.height = 40
        self.center_x = x
        self.center_y = y
        self.change_x = ENEMY_SPEED

        self.physics = arcade.PhysicsEnginePlatformer(
            self, platforms, gravity_constant=GRAVITY
        )

    def update_enemy(self):
        self.physics.update()
        self.center_x += self.change_x

        if not self.physics.can_jump():
            self.change_x *= -1


# --------------------
# АТАКА
# --------------------
class Attack(arcade.Sprite):
    def __init__(self, x, y):
        super().__init__()
        self.texture = arcade.make_soft_square_texture(
            32, arcade.color.YELLOW, 255, 255
        )
        self.width = 30
        self.height = 30
        self.center_x = x
        self.center_y = y
        self.life_time = ATTACK_TIME


# --------------------
# ИГРОВОЕ ОКНО
# --------------------
class GameWindow(arcade.Window):

    def __init__(self):
        super().__init__(SCREEN_WIDTH, SCREEN_HEIGHT, SCREEN_TITLE)
        arcade.set_background_color(arcade.color.DARK_SLATE_GRAY)

        self.camera = arcade.camera.Camera2D()

        self.player_list = arcade.SpriteList()
        self.platforms = arcade.SpriteList(use_spatial_hash=True)
        self.enemies = arcade.SpriteList()
        self.attacks = arcade.SpriteList()

        self.player = None
        self.physics_engine = None

        self.game_over = False

        # ТЕКСТ
        self.hp_text = arcade.Text(
            "", 20, SCREEN_HEIGHT - 30,
            arcade.color.WHITE, 20
        )

        self.game_over_text = arcade.Text(
            "GAME OVER",
            SCREEN_WIDTH // 2,
            SCREEN_HEIGHT // 2 + 40,
            arcade.color.RED,
            60,
            anchor_x="center"
        )

        self.restart_text = arcade.Text(
            "Press R to restart",
            SCREEN_WIDTH // 2,
            SCREEN_HEIGHT // 2 - 20,
            arcade.color.WHITE,
            20,
            anchor_x="center"
        )

    # --------------------
    # SETUP
    # --------------------
    def setup(self):
        self.game_over = False

        self.player_list.clear()
        self.platforms.clear()
        self.enemies.clear()
        self.attacks.clear()

        # Игрок
        self.player = Player()
        self.player.center_x = 400
        self.player.center_y = 150
        self.player_list.append(self.player)

        # Земля
        ground = make_sprite(LEVEL_WIDTH, 40, arcade.color.GRAY)
        ground.center_x = LEVEL_WIDTH // 2
        ground.center_y = 20
        self.platforms.append(ground)

        # Платформы
        for i in range(4):
            p = make_sprite(200, 20, arcade.color.GRAY)
            p.center_x = 500 + i * 350
            p.center_y = 220
            self.platforms.append(p)

        # Враги
        self.enemies.append(Enemy(600, 260, self.platforms))
        self.enemies.append(Enemy(1100, 260, self.platforms))

        self.physics_engine = arcade.PhysicsEnginePlatformer(
            self.player,
            self.platforms,
            gravity_constant=GRAVITY
        )

        self.update_camera()

    # --------------------
    # КАМЕРА
    # --------------------
    def update_camera(self):
        self.camera.position = (
            self.player.center_x - SCREEN_WIDTH / 60,
            self.player.center_y - SCREEN_HEIGHT / 60
        )

    # --------------------
    # DRAW
    # --------------------
    def on_draw(self):
        self.clear()
        self.camera.use()

        self.platforms.draw()
        self.enemies.draw()
        self.attacks.draw()
        self.player_list.draw()

        # UI
        arcade.camera.Camera2D().use()

        self.hp_text.text = f"HP: {self.player.hp}"
        self.hp_text.draw()

        if self.game_over:
            self.game_over_text.draw()
            self.restart_text.draw()

    # --------------------
    # UPDATE
    # --------------------
    def on_update(self, delta_time):
        if self.game_over:
            return

        self.physics_engine.update()

        if self.player.invincible_timer > 0:
            self.player.invincible_timer -= delta_time

        for enemy in self.enemies:
            enemy.update_enemy()

        for attack in self.attacks:
            attack.life_time -= delta_time
            if attack.life_time <= 0:
                attack.remove_from_sprite_lists()

        for attack in self.attacks:
            hit_list = arcade.check_for_collision_with_list(
                attack, self.enemies
            )
            for enemy in hit_list:
                enemy.remove_from_sprite_lists()
                attack.remove_from_sprite_lists()

        if self.player.invincible_timer <= 0:
            hits = arcade.check_for_collision_with_list(
                self.player, self.enemies
            )
            if hits:
                self.player.hp -= 5
                self.player.invincible_timer = INVINCIBLE_TIME

                if self.player.hp <= 0:
                    self.game_over = True

        self.update_camera()

    # --------------------
    # INPUT
    # --------------------
    def on_key_press(self, key, modifiers):
        if self.game_over:
            if key == arcade.key.R:
                self.setup()
            return

        if key == arcade.key.A:
            self.player.change_x = -PLAYER_SPEED
            self.player.facing = -1
        elif key == arcade.key.D:
            self.player.change_x = PLAYER_SPEED
            self.player.facing = 1
        elif key == arcade.key.W:
            if self.physics_engine.can_jump():
                self.player.change_y = JUMP_SPEED
        elif key == arcade.key.SPACE:
            self.spawn_attack()

    def on_key_release(self, key, modifiers):
        if key in (arcade.key.A, arcade.key.D):
            self.player.change_x = 0

    # --------------------
    # АТАКА
    # --------------------
    def spawn_attack(self):
        offset = 35 * self.player.facing
        attack = Attack(
            self.player.center_x + offset,
            self.player.center_y
        )
        self.attacks.append(attack)


# --------------------
# ЗАПУСК
# --------------------
def main():
    window = GameWindow()
    window.setup()
    arcade.run()


if __name__ == "__main__":
    main()
